  <script>
    const STORAGE_KEY = 'gastro_toolkit_v1';

    const defaultData = {
      ingredients: [
        { id: '1', name: 'Hovězí maso zadní', category: 'Maso', unit: 'g', stock: 4500, min: 1500, pricePerUnit: 0.30, exp: getFutureDate(4) },
        { id: '2', name: 'Rýže jasmínová', category: 'Suché', unit: 'g', stock: 8000, min: 2000, pricePerUnit: 0.05, exp: getFutureDate(120) },
        { id: '3', name: 'Čerstvá zelenina mix', category: 'Zelenina', unit: 'g', stock: 900, min: 1200, pricePerUnit: 0.15, exp: getFutureDate(1) },
        { id: '4', name: 'Demi-glace omáčka', category: 'Ostatní', unit: 'ml', stock: 2200, min: 500, pricePerUnit: 0.26, exp: getFutureDate(14) },
        { id: '5', name: 'Smetana 33%', category: 'Mléčné', unit: 'ml', stock: 3000, min: 1000, pricePerUnit: 0.09, exp: getFutureDate(2) }
      ],
      recipes: [
        {
          id: 'r1',
          name: 'Hovězí na zelenině s rýží',
          sellingPrice: 250,
          allergens: '9',
          items: [
            { ingredientId: '1', qty: 150 },
            { ingredientId: '2', qty: 100 },
            { ingredientId: '3', qty: 80 },
            { ingredientId: '4', qty: 30 }
          ]
        }
      ],
      waste: [
        { id: 'w1', ingredientName: 'Čerstvá zelenina mix', qty: 400, unit: 'g', loss: 60, reason: 'Prošlá expirace', date: 'Dnes' }
      ]
    };

    let state = loadState();
    let tempRecipeItems = [];

    function getFutureDate(days) {
      const d = new Date();
      d.setDate(d.getDate() + days);
      return d.toISOString().split('T')[0];
    }

    function loadState() {
      const raw = localStorage.getItem(STORAGE_KEY);
      return raw ? JSON.parse(raw) : JSON.parse(JSON.stringify(defaultData));
    }

    function saveState() {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
      renderAll();
    }

    function resetDemoData() {
      state = JSON.parse(JSON.stringify(defaultData));
      saveState();
      alert('Ukázková data byla obnovena.');
    }

    function switchTab(viewId, btn) {
      document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));
      document.getElementById('view-' + viewId).classList.add('active');
      document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
    }

    function openModal(id) {
      populateSelects();
      document.getElementById(id).classList.add('open');
    }

    function closeModal(id) {
      document.getElementById(id).classList.remove('open');
    }

    function populateSelects() {
      const options = state.ingredients.map(i =>
        `<option value="${i.id}">${i.name} (${i.stock} ${i.unit})</option>`
      ).join('');
      document.getElementById('waste-ingredient').innerHTML = options;
      document.getElementById('rec-ing-select').innerHTML = options;
    }

    function calcRecipeCost(recipe) {
      return recipe.items.reduce((sum, item) => {
        const ing = state.ingredients.find(i => i.id === item.ingredientId);
        return sum + (ing ? ing.pricePerUnit * item.qty : 0);
      }, 0);
    }

    function renderAll() {
      // KPI Sklad
      const stockValue = state.ingredients.reduce((s, i) => s + (i.stock * i.pricePerUnit), 0);
      document.getElementById('kpi-stock').textContent = Math.round(stockValue).toLocaleString('cs-CZ') + ' Kč';
      document.getElementById('kpi-items-count').textContent = state.ingredients.length + ' položek';

      // KPI Odpisy
      const wasteValue = state.waste.reduce((s, w) => s + w.loss, 0);
      document.getElementById('kpi-waste').textContent = Math.round(wasteValue).toLocaleString('cs-CZ') + ' Kč';
      document.getElementById('waste-total-big').textContent = Math.round(wasteValue).toLocaleString('cs-CZ') + ' Kč';
      document.getElementById('kpi-waste-count').textContent = state.waste.length + ' záznamů';

      // KPI Receptury & Food Cost
      document.getElementById('kpi-recipes').textContent = state.recipes.length;
      let avgFc = 0;
      if (state.recipes.length > 0) {
        const totalFc = state.recipes.reduce((s, r) => {
          const cost = calcRecipeCost(r);
          return s + (r.sellingPrice > 0 ? (cost / r.sellingPrice) * 100 : 0);
        }, 0);
        avgFc = totalFc / state.recipes.length;
      }
      document.getElementById('kpi-foodcost').textContent = avgFc.toFixed(1) + ' %';

      // Upozornění
      const alerts = [];
      const today = new Date().toISOString().split('T')[0];
      const inTwoDays = getFutureDate(2);

      state.ingredients.forEach(i => {
        if (i.stock <= i.min) {
          alerts.push({ text: `Nízká zásoba: ${i.name} (${i.stock} ${i.unit})`, type: 'warn' });
        }
        if (i.exp && i.exp <= inTwoDays) {
          alerts.push({ text: `Blíží se expirace: ${i.name} (${i.exp})`, type: 'danger' });
        }
      });

      document.getElementById('alerts-badge').textContent = alerts.length;
      document.getElementById('dashboard-alerts').innerHTML = alerts.length
        ? alerts.map(a => `<div class="alert-item ${a.type === 'danger' ? 'danger' : ''}">${a.text}</div>`).join('')
        : '<div style="font-size:13px; color:var(--muted);">Vše v pořádku, žádné kritické stavy.</div>';

      // Kategorie graf
      const cats = {};
      state.ingredients.forEach(i => {
        cats[i.category] = (cats[i.category] || 0) + (i.stock * i.pricePerUnit);
      });
      document.getElementById('category-bars').innerHTML = Object.entries(cats).map(([cat, val]) => {
        const pct = stockValue > 0 ? Math.round((val / stockValue) * 100) : 0;
        return `
          <div style="margin-top:10px;">
            <div style="display:flex; justify-content:space-between; font-size:12px;">
              <span>${cat}</span>
              <span>${Math.round(val)} Kč (${pct} %)</span>
            </div>
            <div class="bar-bg"><div class="bar-fill" style="width:${pct}%"></div></div>
          </div>`;
      }).join('');

      renderRecipes();
      renderStock();
      renderWaste();
      renderAdvisor(avgFc, wasteValue);
    }

    function renderRecipes() {
      document.getElementById('recipes-list').innerHTML = state.recipes.map(r => {
        const cost = calcRecipeCost(r);
        const fc = r.sellingPrice > 0 ? (cost / r.sellingPrice) * 100 : 0;
        const margin = r.sellingPrice - cost;
        const fcClass = fc > 35 ? 'bad' : (fc > 30 ? 'warn' : '');

        const ingList = r.items.map(it => {
          const ing = state.ingredients.find(i => i.id === it.ingredientId);
          return ing ? `${ing.name} (${it.qty} ${ing.unit})` : '';
        }).join(', ');

        return `
          <div class="card">
            <div style="display:flex; justify-content:space-between; align-items:flex-start; margin-bottom:8px;">
              <div>
                <strong style="font-size:16px;">${r.name}</strong>
                <div style="font-size:12px; color:var(--muted); margin-top:2px;">Alergeny: ${r.allergens || 'Neuvedeno'}</div>
              </div>
              <span class="badge ${fcClass}">FC ${fc.toFixed(1)} %</span>
            </div>
            <div style="font-size:12px; color:var(--muted); margin-bottom:10px;">${ingList}</div>
            <div style="display:grid; grid-template-columns: repeat(3, 1fr); gap:8px; background:var(--surface-2); padding:10px; border-radius:10px; font-size:12px;">
              <div><span style="color:var(--muted);">Náklad:</span><br><strong>${cost.toFixed(1)} Kč</strong></div>
              <div><span style="color:var(--muted);">Prodej:</span><br><strong>${r.sellingPrice} Kč</strong></div>
              <div><span style="color:var(--muted);">Rozdíl:</span><br><strong class="text-success">${margin.toFixed(1)} Kč</strong></div>
            </div>
          </div>`;
      }).join('');
    }

    function renderStock() {
      const q = (document.getElementById('stock-search').value || '').toLowerCase();
      const filtered = state.ingredients.filter(i => i.name.toLowerCase().includes(q));

      document.getElementById('stock-list').innerHTML = filtered.map(i => {
        const low = i.stock <= i.min;
        return `
          <div class="list-item">
            <div>
              <strong>${i.name}</strong>
              <div style="font-size:12px; color:var(--muted);">${i.category} · Exp: ${i.exp || '—'}</div>
            </div>
            <div style="text-align:right;">
              <strong class="${low ? 'text-warning' : ''}">${i.stock} ${i.unit}</strong>
              <div style="font-size:11px; color:var(--muted);">${Math.round(i.stock * i.pricePerUnit)} Kč</div>
            </div>
          </div>`;
      }).join('') || '<div style="font-size:13px; color:var(--muted);">Žádné suroviny nenalezeny.</div>';
    }

    function renderWaste() {
      document.getElementById('waste-list').innerHTML = state.waste.slice().reverse().map(w => `
        <div class="list-item">
          <div>
            <strong>${w.ingredientName}</strong>
            <div style="font-size:12px; color:var(--muted);">${w.reason} · ${w.date}</div>
          </div>
          <div style="text-align:right;">
            <strong class="text-danger">-${Math.round(w.loss)} Kč</strong>
            <div style="font-size:11px; color:var(--muted);">${w.qty} ${w.unit}</div>
          </div>
        </div>
      `).join('');
    }

    function renderAdvisor(avgFc, wasteValue) {
      const tips = [];
      if (avgFc > 30) {
        tips.push(`Průměrný food cost je ${avgFc.toFixed(1)} %. Zkontroluj prodejní ceny nebo gramáže u nejdražších surovin.`);
      } else {
        tips.push(`Průměrný food cost (${avgFc.toFixed(1)} %) držíš pod hranicí 30 %, což je zdravý základ provozu.`);
      }
      if (wasteValue > 0) {
        tips.push(`Evidované odpisy dosáhly ${Math.round(wasteValue)} Kč. Sleduj především suroviny s krátkou expirací.`);
      }
      const lowItems = state.ingredients.filter(i => i.stock <= i.min);
      if (lowItems.length > 0) {
        tips.push(`Do nákupního seznamu doporučuji ihned zařadit: ${lowItems.map(i => i.name).join(', ')}.`);
      }

      document.getElementById('advisor-insights').innerHTML = tips.map(t =>
        `<div class="card" style="border-left:3px solid var(--primary); font-size:13px; line-height:1.5;">${t}</div>`
      ).join('');
    }

    // Akce formulářů
    function submitWaste() {
      const id = document.getElementById('waste-ingredient').value;
      const qty = parseFloat(document.getElementById('waste-qty').value);
      const reason = document.getElementById('waste-reason').value;
      if (!id || !qty || qty <= 0) return alert('Zadej platné množství.');

      const ing = state.ingredients.find(i => i.id === id);
      const loss = qty * ing.pricePerUnit;
      ing.stock = Math.max(0, ing.stock - qty);

      state.waste.push({
        id: Date.now().toString(),
        ingredientName: ing.name,
        qty,
        unit: ing.unit,
        loss,
        reason,
        date: new Date().toLocaleDateString('cs-CZ')
      });

      document.getElementById('waste-qty').value = '';
      closeModal('wasteModal');
      saveState();
    }

    function submitStock() {
      const name = document.getElementById('st-name').value.trim();
      const category = document.getElementById('st-cat').value;
      const unit = document.getElementById('st-unit').value;
      const qty = parseFloat(document.getElementById('st-qty').value);
      const totalPrice = parseFloat(document.getElementById('st-total-price').value);
      const min = parseFloat(document.getElementById('st-min').value) || 0;
      const exp = document.getElementById('st-exp').value;

      if (!name || !qty || !totalPrice) return alert('Vyplň název, množství a cenu.');

      const pricePerUnit = totalPrice / qty;
      const existing = state.ingredients.find(i => i.name.toLowerCase() === name.toLowerCase());

      if (existing) {
        existing.stock += qty;
        existing.pricePerUnit = pricePerUnit;
        if (exp) existing.exp = exp;
      } else {
        state.ingredients.push({
          id: Date.now().toString(),
          name,
          category,
          unit,
          stock: qty,
          min,
          pricePerUnit,
          exp
        });
      }

      document.getElementById('st-name').value = '';
      document.getElementById('st-qty').value = '';
      document.getElementById('st-total-price').value = '';
      closeModal('stockModal');
      saveState();
    }

    function addTempIngredient() {
      const ingredientId = document.getElementById('rec-ing-select').value;
      const qty = parseFloat(document.getElementById('rec-ing-qty').value);
      if (!ingredientId || !qty || qty <= 0) return;

      tempRecipeItems.push({ ingredientId, qty });
      document.getElementById('rec-ing-qty').value = '';
      renderTempRecipeItems();
    }

    function renderTempRecipeItems() {
      document.getElementById('temp-recipe-ings').innerHTML = tempRecipeItems.map(it => {
        const ing = state.ingredients.find(i => i.id === it.ingredientId);
        return `<div>• ${ing.name}: ${it.qty} ${ing.unit} (${(ing.pricePerUnit * it.qty).toFixed(1)} Kč)</div>`;
      }).join('');
    }

    function submitRecipe() {
      const name = document.getElementById('rec-name').value.trim();
      const sellingPrice = parseFloat(document.getElementById('rec-price').value);
      const allergens = document.getElementById('rec-allergens').value.trim();

      if (!name || !sellingPrice || tempRecipeItems.length === 0) {
        return alert('Vyplň název, prodejní cenu a přidej alespoň jednu surovinu.');
      }

      state.recipes.push({
        id: Date.now().toString(),
        name,
        sellingPrice,
        allergens,
        items: [...tempRecipeItems]
      });

      tempRecipeItems = [];
      document.getElementById('rec-name').value = '';
      document.getElementById('rec-price').value = '';
      document.getElementById('temp-recipe-ings').innerHTML = '';
      closeModal('recipeModal');
      saveState();
    }

    renderAll();
  </script>
